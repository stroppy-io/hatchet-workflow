# Agent Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the agent subsystem — server-side `AgentService` + `AgentAdminService`, the in-memory `Hub` that bridges DAG node handlers and connected agents, the `agent_commands` persist-and-replay table, and the agent-side daemon binary that implements the 5 verbs (`PutFile`, `RunShell`, `SystemctlOp`, `InstallPackage`, `WaitForCondition`).

**Architecture:** Wire protocol matches `protocols/cloud/v1/agent/service.proto` verbatim — unary `Register` → unary `Poll` (long-poll: server holds connection up to ~30 s waiting for new commands) → unary `Deregister`. The `PollRequest` carries `AgentReport` (heartbeat + completed reports + log batches) and the response is a `CommandBatch` (pending verbs to execute). Server side: `agent.Hub` indexes per-agent command queues and pending-reply waiters; node handlers enqueue commands via `Hub.Dispatch` and block on a per-command channel until the agent's next `Poll` posts a `Report` for that command id. Persistence: every dispatched command is `INSERT`ed into `agent_commands` (state PENDING → DELIVERED → REPORTED), so on server restart a node handler can detect already-reported commands and not re-issue.

Agent binary: Cobra subcommand `stroppy-cloud agent`. Reads server URL + bootstrap_token from local file or cloud-init. On start: `Register` → persist `agent_id`+`poll_token` → enter Poll loop with reconnect backoff. Per-verb handlers run with `exec.CommandContext`; cancellation flows from server via the next `Poll` returning a `Cancel` directive.

**Tech Stack:** ConnectRPC unary, ratel `ProtoRepository` on `agent_commands` + `agents`, avito-tech `pgtx`, `os/exec`, `apt-get`/`dpkg`/`tar` binary install paths, `systemctl` shell out, `internal/infrastructure/s3` for `FetchSpec.url=s3://...`, `internal/infrastructure/valkey` for bootstrap-token cache.

**Prerequisites:** Plans 01-04 merged. The `system.NodeHandler` set from Plan 03 currently dispatches via a stub `Hub`; this plan replaces the stub with the real Hub and real DB persistence.

package agent

import (
	"context"

	"github.com/arenadata/stroppy-io/stroppy-cloud/internal/core/domainerr"
	errorspb "github.com/arenadata/stroppy-io/stroppy-cloud/internal/proto/cloud/v1/errors"
	agentpb "github.com/arenadata/stroppy-io/stroppy-cloud/internal/proto/cloud/v1/agent"
	"github.com/naukograd-software/ratel/pkg/exec"
	"github.com/naukograd-software/ratel/pkg/repository"
)

type CommandsRepo struct {
	repo *repository.ProtoRepository[agentpb.AgentCommandAlias, agentpb.AgentCommandColumnAlias, *agentpb.AgentCommandScanner, *agentpb.AgentCommand]
}

func NewCommandsRepo(executor exec.DB) *CommandsRepo {
	return &CommandsRepo{
		repo: repository.NewProtoRepository(
			repository.NewScannerRepository(agentpb.AgentCommands.Table, executor),
			agentpb.AgentCommandConverter,
		),
	}
}

func (r *CommandsRepo) InsertPending(ctx context.Context, cmd *agentpb.AgentCommand) error {
	cmd.State = agentpb.AgentCommandState_AGENT_COMMAND_STATE_PENDING
	cmd.IssuedAt = timestamppb.Now()
	return domainerr.FromPg(r.repo.Insert(ctx, cmd))
}

func (r *CommandsRepo) MarkDelivered(ctx context.Context, id string) error {
	return r.transition(ctx, id, agentpb.AgentCommandState_AGENT_COMMAND_STATE_PENDING, agentpb.AgentCommandState_AGENT_COMMAND_STATE_DELIVERED, func(c *agentpb.AgentCommand) {
		c.DeliveredAt = timestamppb.Now()
	})
}

func (r *CommandsRepo) MarkReported(ctx context.Context, id string, payload []byte) error {
	return r.transition(ctx, id, agentpb.AgentCommandState_AGENT_COMMAND_STATE_DELIVERED, agentpb.AgentCommandState_AGENT_COMMAND_STATE_REPORTED, func(c *agentpb.AgentCommand) {
		c.ReportedAt = timestamppb.Now()
		c.ResultPayload = payload
	})
}

func (r *CommandsRepo) MarkTimedOut(ctx context.Context, id string) error {
	return r.transition(ctx, id, agentpb.AgentCommandState_AGENT_COMMAND_STATE_DELIVERED, agentpb.AgentCommandState_AGENT_COMMAND_STATE_TIMED_OUT, nil)
}

func (r *CommandsRepo) FindByNodeRunID(ctx context.Context, nrID *systempb.NodeRunId) (*agentpb.AgentCommand, error) {
	rows, err := r.repo.FindAll(ctx, agentpb.AgentCommandColumnAlias.NodeRunId.EqQ(nrID.GetValue()))
	if err != nil { return nil, domainerr.FromPg(err) }
	if len(rows) == 0 { return nil, domainerr.E(errorspb.Code_NOT_FOUND) }
	return rows[0], nil
}

func (r *CommandsRepo) transition(ctx context.Context, id string, from, to agentpb.AgentCommandState, mutate func(*agentpb.AgentCommand)) error {
	cmd, err := r.repo.FindByID(ctx, id)
	if err != nil { return domainerr.FromPg(err) }
	if cmd.State != from {
		return domainerr.E(errorspb.Code_FAILED_PRECONDITION)
	}
	cmd.State = to
	if mutate != nil { mutate(cmd) }
	return domainerr.FromPg(r.repo.Update(ctx, cmd))
}

func (r *CommandsRepo) ClaimForAgent(ctx context.Context, agentID *agentpb.AgentId, limit int) ([]*agentpb.AgentCommand, error) {
	// SELECT ... WHERE agent_id=? AND state IN (PENDING, DELIVERED) ORDER BY issued_at LIMIT ?
	// FOR UPDATE SKIP LOCKED to allow concurrent Polls if any.
	return r.repo.FindAllRaw(ctx, claimSQL, agentID.GetValue(), limit)
}

const claimSQL = `SELECT * FROM agent_commands WHERE agent_id=$1 AND state IN ('AGENT_COMMAND_STATE_PENDING','AGENT_COMMAND_STATE_DELIVERED') ORDER BY issued_at LIMIT $2 FOR UPDATE SKIP LOCKED`
```

package agent

import (
	"context"
	"sync"
	"time"

	"google.golang.org/protobuf/proto"
	"github.com/arenadata/stroppy-io/stroppy-cloud/internal/core/domainerr"
	errorspb "github.com/arenadata/stroppy-io/stroppy-cloud/internal/proto/cloud/v1/errors"
	agentpb "github.com/arenadata/stroppy-io/stroppy-cloud/internal/proto/cloud/v1/agent"
	systempb "github.com/arenadata/stroppy-io/stroppy-cloud/internal/proto/cloud/v1/system"
)

type Hub struct {
	mu       sync.Mutex
	queues   map[string]chan *agentpb.Command // agent_id → pending commands
	pending  map[string]chan *agentpb.CommandReport
	repo     *CommandsRepo
}

func NewHub(repo *CommandsRepo) *Hub {
	return &Hub{
		queues:  map[string]chan *agentpb.Command{},
		pending: map[string]chan *agentpb.CommandReport{},
		repo:    repo,
	}
}

func (h *Hub) DispatchForNodeRun(ctx context.Context, agentID *agentpb.AgentId, nodeRunID *systempb.NodeRunId, cmd *agentpb.Command, timeout time.Duration) (*agentpb.CommandReport, error) {
	if existing, err := h.repo.FindByNodeRunID(ctx, nodeRunID); err == nil &&
		existing.State == agentpb.AgentCommandState_AGENT_COMMAND_STATE_REPORTED {
		rep := &agentpb.CommandReport{}
		if err := proto.Unmarshal(existing.ResultPayload, rep); err != nil {
			return nil, err
		}
		return rep, nil
	}

	payload, err := proto.Marshal(cmd)
	if err != nil { return nil, err }
	if err := h.repo.InsertPending(ctx, &agentpb.AgentCommand{
		Id: cmd.Id, AgentId: agentID, NodeRunId: nodeRunID, CommandPayload: payload,
	}); err != nil {
		return nil, err
	}
	return h.Dispatch(ctx, agentID, cmd, timeout)
}

func (h *Hub) Dispatch(ctx context.Context, agentID *agentpb.AgentId, cmd *agentpb.Command, timeout time.Duration) (*agentpb.CommandReport, error) {
	replyCh := make(chan *agentpb.CommandReport, 1)
	h.mu.Lock()
	q := h.queueFor(agentID.GetValue())
	h.pending[cmd.Id] = replyCh
	h.mu.Unlock()

	select {
	case q <- cmd:
	case <-ctx.Done():
		h.forgetPending(cmd.Id)
		return nil, ctx.Err()
	}

	timer := time.NewTimer(timeout)
	defer timer.Stop()
	select {
	case rep := <-replyCh:
		return rep, nil
	case <-timer.C:
		h.forgetPending(cmd.Id)
		_ = h.repo.MarkTimedOut(ctx, cmd.Id)
		return nil, domainerr.E(errorspb.Code_DEADLINE_EXCEEDED)
	case <-ctx.Done():
		h.forgetPending(cmd.Id)
		return nil, ctx.Err()
	}
}

// Drain — called by Poll handler. Returns up to `limit` commands. Marks delivered in DB.
func (h *Hub) Drain(ctx context.Context, agentID *agentpb.AgentId, limit int) []*agentpb.Command {
	h.mu.Lock()
	q := h.queueFor(agentID.GetValue())
	h.mu.Unlock()

	out := []*agentpb.Command{}
	for len(out) < limit {
		select {
		case cmd := <-q:
			out = append(out, cmd)
			_ = h.repo.MarkDelivered(ctx, cmd.Id)
		default:
			return out
		}
	}
	return out
}

// WaitForCommand — long-poll variant: if queue empty, wait up to `timeout` for first command.
func (h *Hub) WaitForCommand(ctx context.Context, agentID *agentpb.AgentId, timeout time.Duration) []*agentpb.Command {
	if cmds := h.Drain(ctx, agentID, 32); len(cmds) > 0 {
		return cmds
	}
	h.mu.Lock(); q := h.queueFor(agentID.GetValue()); h.mu.Unlock()
	t := time.NewTimer(timeout); defer t.Stop()
	select {
	case cmd := <-q:
		_ = h.repo.MarkDelivered(ctx, cmd.Id)
		// drain any additional ready commands
		extra := h.Drain(ctx, agentID, 31)
		return append([]*agentpb.Command{cmd}, extra...)
	case <-t.C:
		return nil
	case <-ctx.Done():
		return nil
	}
}

func (h *Hub) Resolve(cmdID string, rep *agentpb.CommandReport) {
	h.mu.Lock(); ch := h.pending[cmdID]; delete(h.pending, cmdID); h.mu.Unlock()
	if ch != nil { ch <- rep }
}

func (h *Hub) queueFor(agentID string) chan *agentpb.Command {
	q, ok := h.queues[agentID]
	if !ok {
		q = make(chan *agentpb.Command, 256)
		h.queues[agentID] = q
	}
	return q
}

func (h *Hub) forgetPending(id string) {
	h.mu.Lock(); delete(h.pending, id); h.mu.Unlock()
}
```

```go
package agent

import (
	"context"
	"time"

	"github.com/golang-jwt/jwt/v5"
	"github.com/arenadata/stroppy-io/stroppy-cloud/internal/core/domainerr"
	"github.com/arenadata/stroppy-io/stroppy-cloud/internal/core/ids"
	"github.com/arenadata/stroppy-io/stroppy-cloud/internal/infrastructure/valkey"
	catalogpb "github.com/arenadata/stroppy-io/stroppy-cloud/internal/proto/cloud/v1/catalog"
	errorspb "github.com/arenadata/stroppy-io/stroppy-cloud/internal/proto/cloud/v1/errors"
	iampb "github.com/arenadata/stroppy-io/stroppy-cloud/internal/proto/cloud/v1/iam"
	systempb "github.com/arenadata/stroppy-io/stroppy-cloud/internal/proto/cloud/v1/system"
)

type BootstrapClaims struct {
	TenantId  *iampb.TenantId
	DagRunId  *systempb.DagRunId
	MachineId string
	Role      catalogpb.MachineRole
	TTL       time.Duration
}

type BootstrapTokenStore struct {
	kv     *valkey.Client
	secret []byte
}

func NewBootstrapTokenStore(kv *valkey.Client, secret []byte) *BootstrapTokenStore {
	return &BootstrapTokenStore{kv: kv, secret: secret}
}

func (s *BootstrapTokenStore) Issue(ctx context.Context, c BootstrapClaims) (string, error) {
	jti := ids.NewULID()
	t := jwt.NewWithClaims(jwt.SigningMethodHS256, jwt.MapClaims{
		"jti":        jti,
		"tenant_id":  c.TenantId.GetValue(),
		"dag_run_id": c.DagRunId.GetValue(),
		"machine_id": c.MachineId,
		"role":       int(c.Role),
		"exp":        time.Now().Add(c.TTL).Unix(),
	})
	signed, err := t.SignedString(s.secret)
	if err != nil { return "", err }
	if err := s.kv.SetWithTTL(ctx, "bs:"+jti, "1", c.TTL); err != nil { return "", err }
	return signed, nil
}

func (s *BootstrapTokenStore) Verify(ctx context.Context, tok string) (BootstrapClaims, error) {
	parsed, err := jwt.Parse(tok, func(t *jwt.Token) (any, error) { return s.secret, nil })
	if err != nil || !parsed.Valid {
		return BootstrapClaims{}, domainerr.E(errorspb.Code_UNAUTHENTICATED)
	}
	mc := parsed.Claims.(jwt.MapClaims)
	jti := mc["jti"].(string)
	ok, err := s.kv.Exists(ctx, "bs:"+jti)
	if err != nil || !ok {
		return BootstrapClaims{}, domainerr.E(errorspb.Code_UNAUTHENTICATED)
	}
	return BootstrapClaims{
		TenantId:  &iampb.TenantId{Value: mc["tenant_id"].(string)},
		DagRunId:  &systempb.DagRunId{Value: mc["dag_run_id"].(string)},
		MachineId: mc["machine_id"].(string),
		Role:      catalogpb.MachineRole(int(mc["role"].(float64))),
	}, nil
}

func (s *BootstrapTokenStore) Invalidate(ctx context.Context, tok string) error {
	parsed, err := jwt.Parse(tok, func(t *jwt.Token) (any, error) { return s.secret, nil })
	if err != nil { return nil }
	jti := parsed.Claims.(jwt.MapClaims)["jti"].(string)
	return s.kv.Delete(ctx, "bs:"+jti)
}

func (s *Service) Register(ctx context.Context, req *agentpb.RegisterRequest) (*agentpb.RegisterResponse, error) {
	return tracing.WithTraceRet(s.Tracer(), ctx, "Register", func(ctx context.Context, _ trace.Span) (*agentpb.RegisterResponse, error) {
		claims, err := s.bootstrap.Verify(ctx, req.BootstrapToken)
		if err != nil { return nil, err }
		if claims.MachineId != req.MachineId || claims.Role != req.Role {
			return nil, domainerr.E(errorspb.Code_PERMISSION_DENIED)
		}
		return pgtx.WithSerializableRet(ctx, s.txManager, func(ctx context.Context) (*agentpb.RegisterResponse, error) {
			ag := &agentpb.Agent{
				Id: &agentpb.AgentId{Value: ids.NewULID()},
				TenantId: claims.TenantId, DagRunId: claims.DagRunId,
				MachineId: req.MachineId, Role: req.Role,
				InternalIp: req.InternalIp, PublicIp: req.PublicIp,
				AgentVersion: req.AgentVersion, Capabilities: req.Capabilities,
				Status: agentpb.AgentStatus_AGENT_STATUS_IDLE,
				RegisteredAt: timestamppb.Now(),
			}
			if err := s.agents.Insert(ctx, ag); err != nil { return nil, domainerr.FromPg(err) }
			pollToken, err := s.authJWT.IssueAgent(ag.Id, claims.TenantId, claims.DagRunId)
			if err != nil { return nil, err }
			_ = s.bootstrap.Invalidate(ctx, req.BootstrapToken)
			return &agentpb.RegisterResponse{Agent: ag, PollToken: pollToken}, nil
		})
	})
}

func (s *Service) Poll(ctx context.Context, req *agentpb.PollRequest) (*agentpb.CommandBatch, error) {
	return tracing.WithTraceRet(s.Tracer(), ctx, "Poll", func(ctx context.Context, _ trace.Span) (*agentpb.CommandBatch, error) {
		authed, err := s.authJWT.AgentFromCtx(ctx)
		if err != nil { return nil, err }
		if authed.GetValue() != req.AgentId.GetValue() {
			return nil, domainerr.E(errorspb.Code_PERMISSION_DENIED)
		}
		if err := s.processReport(ctx, req.AgentId, req.Report); err != nil { return nil, err }
		// long-poll up to 25s (below typical 30s reverse-proxy idle timeout)
		cmds := s.hub.WaitForCommand(ctx, req.AgentId, 25*time.Second)
		return &agentpb.CommandBatch{Commands: cmds, PollDelayMs: 0}, nil
	})
}

func (s *Service) processReport(ctx context.Context, agentID *agentpb.AgentId, r *agentpb.AgentReport) error {
	if r == nil { return nil }
	if r.Heartbeat != nil {
		_ = s.agents.UpdateField(ctx, agentID.GetValue(), "last_seen_at", time.Now())
	}
	for _, cr := range r.Commands {
		payload, _ := proto.Marshal(cr)
		_ = s.commands.MarkReported(ctx, cr.CommandId, payload)
		s.hub.Resolve(cr.CommandId, cr)
	}
	for _, batch := range r.Logs {
		_ = s.logBus.Publish(ctx, batch) // also INSERT into node_run_logs partitioned table
	}
	return nil
}

package daemon

type Config struct {
	ServerURL       string
	BootstrapToken  string
	MachineID       string
	StateDir        string
	PollTimeout     time.Duration
	IdempotencyCap  int
	AgentVersion    string
}

type Daemon struct {
	cfg     Config
	cli     agentv1connect.AgentServiceClient
	state   *State           // persisted: agent_id, poll_token
	dispatch *Dispatcher
	cache   *IdempotencyCache
	logger  *zap.Logger
}

func New(cfg Config) *Daemon { /* construct, no IO */ return &Daemon{cfg: cfg, dispatch: NewDispatcher(), cache: NewIdempotencyCache(cfg.StateDir, cfg.IdempotencyCap)} }

func (d *Daemon) Run(ctx context.Context) error {
	if err := d.ensureRegistered(ctx); err != nil { return err }
	for {
		if err := d.pollOnce(ctx); err != nil {
			if errors.Is(err, context.Canceled) { return d.deregister(ctx) }
			d.logger.Warn("poll failed", zap.Error(err))
			select { case <-time.After(backoff()): case <-ctx.Done(): return d.deregister(ctx) }
			continue
		}
	}
}

func (d *Daemon) ensureRegistered(ctx context.Context) error {
	if d.state.AgentID != "" { return nil }
	resp, err := d.cli.Register(ctx, connect.NewRequest(&agentpb.RegisterRequest{
		BootstrapToken: d.cfg.BootstrapToken, MachineId: d.cfg.MachineID,
		Role: d.cfg.Role, InternalIp: detectInternalIP(), AgentVersion: d.cfg.AgentVersion,
	}))
	if err != nil { return err }
	d.state.AgentID = resp.Msg.Agent.Id.Value
	d.state.PollToken = resp.Msg.PollToken
	return d.state.Save()
}

func (d *Daemon) pollOnce(ctx context.Context) error {
	report := d.dispatch.DrainReports() // collects completed CommandReports from worker goroutines + LogBatches + heartbeat
	req := connect.NewRequest(&agentpb.PollRequest{
		AgentId: &agentpb.AgentId{Value: d.state.AgentID},
		Report:  report,
	})
	req.Header().Set("Authorization", "Bearer "+d.state.PollToken)

	batch, err := d.cli.Poll(ctx, req)
	if err != nil { return err }
	for _, cmd := range batch.Msg.Commands {
		if cached := d.cache.LookupReport(cmd.Id); cached != nil {
			d.dispatch.QueueReport(cached); continue
		}
		go d.runCommand(ctx, cmd)
	}
	return nil
}

func (d *Daemon) runCommand(ctx context.Context, cmd *agentpb.Command) {
	if cmd.MachineId != d.cfg.MachineID {
		d.dispatch.QueueReport(failed(cmd.Id, "machine_id mismatch"))
		return
	}
	rep := d.dispatch.Dispatch(ctx, cmd)
	d.cache.Store(cmd.Id, rep)
	d.dispatch.QueueReport(rep)
}
```

`Dispatcher` holds a map `verb_kind → VerbHandler`; `Dispatch` switches on `cmd.Action.Verb` and calls the handler, returns `*CommandReport`. `DrainReports` returns a fresh `AgentReport` carrying queued reports + heartbeat + log lines from all in-flight verbs.

package verbs

import (
	"bufio"
	"context"
	"errors"
	"io"
	"os/exec"
	"time"

	"go.uber.org/zap"
	agentpb "github.com/arenadata/stroppy-io/stroppy-cloud/internal/proto/cloud/v1/agent"
)

type RunShellHandler struct{ log *zap.Logger }

func NewRunShellHandler(log *zap.Logger) *RunShellHandler { return &RunShellHandler{log: log} }

func (h *RunShellHandler) Run(ctx context.Context, cmd *agentpb.Command, sink LogSinkFn) (*agentpb.CommandReport, error) {
	rs := cmd.GetAction().GetRunShell()
	if rs == nil { return nil, errors.New("not RunShell") }

	timeout := time.Duration(rs.TimeoutSeconds) * time.Second
	if timeout == 0 { timeout = 1 * time.Hour }
	cctx, cancel := context.WithTimeout(ctx, timeout)
	defer cancel()

	var argv []string
	if rs.Shell {
		argv = []string{"/bin/sh", "-c", joinShell(rs.Argv)}
	} else {
		argv = rs.Argv
	}
	c := exec.CommandContext(cctx, argv[0], argv[1:]...)
	c.Env = envSlice(rs.Env)
	c.Dir = rs.Cwd
	if rs.Stdin != "" { c.Stdin = strings.NewReader(rs.Stdin) }
	stdout, _ := c.StdoutPipe()
	stderr, _ := c.StderrPipe()

	if err := c.Start(); err != nil {
		return failed(cmd.Id, err.Error()), nil
	}
	go pumpLogs(cmd.Id, "stdout", stdout, sink)
	go pumpLogs(cmd.Id, "stderr", stderr, sink)

	err := c.Wait()
	exit := c.ProcessState.ExitCode()
	if cctx.Err() == context.DeadlineExceeded {
		return &agentpb.CommandReport{CommandId: cmd.Id, Outcome: agentpb.Outcome_OUTCOME_TIMEOUT,
			Result: &agentpb.CommandReport_RunShell{RunShell: &agentpb.RunShellResult{ExitCode: int32(exit)}}}, nil
	}
	allowed := contains(rs.ExpectedExits, int32(exit)) || (len(rs.ExpectedExits) == 0 && exit == 0)
	outcome := agentpb.Outcome_OUTCOME_SUCCESS
	if !allowed { outcome = agentpb.Outcome_OUTCOME_FAILED }
	return &agentpb.CommandReport{CommandId: cmd.Id, Outcome: outcome,
		Result: &agentpb.CommandReport_RunShell{RunShell: &agentpb.RunShellResult{ExitCode: int32(exit)}}}, nil
}

func pumpLogs(cmdID, stream string, r io.Reader, sink LogSinkFn) {
	s := bufio.NewScanner(r)
	for s.Scan() {
		sink(&agentpb.LogLine{CommandId: cmdID, Stream: stream, Line: s.Text(), TsMs: time.Now().UnixMilli()})
	}
}

type InstallPackageHandler struct {
	hub      *agentsvc.Hub
	agents   AgentLookup // (dagRunID, role) → agentID
	pkgs     catalogsvc.PackagePort
}

func (h *InstallPackageHandler) Execute(ctx context.Context, nr *systempb.NodeRun, spec *tasksp.InstallPackageTask) error {
	pkg, err := h.pkgs.Get(ctx, spec.PackageId)
	if err != nil { return err }
	targets, err := h.agents.Resolve(ctx, nr.DagRunId, spec.TargetHosts)
	if err != nil { return err }

	for _, ag := range targets {
		cmd := &agentpb.Command{
			Id: ids.NewULID(), MachineId: ag.MachineId, RunId: nr.DagRunId.GetValue(),
			Action: &agentpb.Action{Verb: &agentpb.Action_InstallPackage{InstallPackage: &agentpb.InstallPackage{
				PackageId: pkg.Id, // agent calls BinaryCache.Resolve to translate
			}}},
		}
		rep, err := h.hub.DispatchForNodeRun(ctx, ag.Id, nr.Id, cmd, 30*time.Minute)
		if err != nil { return err }
		if rep.Outcome != agentpb.Outcome_OUTCOME_SUCCESS {
			return domainerr.E(errorspb.Code_INTERNAL,
				&errorspb.Detail{Kind: &errorspb.Detail_ResourceInfo{
					ResourceInfo: &errorspb.ResourceInfo{ResourceType: "agent_command", ResourceName: cmd.Id}}})
		}
	}
	return nil
}
