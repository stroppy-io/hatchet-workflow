# Frontend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Migrate the SPA at `stroppy-cloud/web/` from the legacy REST + WebSocket client (`src/api/{client.ts,ws.ts}`) to ConnectRPC against the new proto services, preserving all existing user flows (login, tenant select, runs, suites, presets, packages, shared run, admin).

**Architecture:** Drop `axios`+`fetch` REST client and bespoke `WebSocket` wrapper. Replace with `@connectrpc/connect-web` transport + generated TypeScript clients from `protoc-gen-es` + `protoc-gen-connect-es`. Server-streams (`WatchTestRun`, `StreamTestRunLogs`, `WatchTestSuiteRun`, `StreamTestSuiteRunLogs`) consumed via `for await` on Connect streams. Auth: access token in memory + refresh token in `httpOnly` cookie set by server on `Login` (cookie is set via Connect response headers); a transport interceptor refreshes on `UNAUTHENTICATED` and retries once. Tenant context selected via `SelectTenant` page → stored in `localStorage` and injected as `X-Tenant-Id` header by a transport interceptor. Routing unchanged (`react-router-dom@7`); page components rewritten to call the new clients. State management stays local (`useState`/`useReducer` + `contexts/`) — no Redux. DAG visualization keeps `@xyflow/react`, fed by `WatchTestRun` progress events instead of WS messages.

**Tech Stack:** Vite + React 19 + TypeScript + Tailwind + Radix + `@connectrpc/connect-web` + `@bufbuild/protobuf`. Build proto codegen via `easyp` (already configured for Go; add JS plugin section).

**Prerequisites:** Plans 01-07 merged. Server endpoints stable.

let accessToken: string | null = null;
let onUnauthenticated: (() => void) | null = null;

export const setAccessToken = (t: string | null) => { accessToken = t; };
export const getAccessToken = () => accessToken;
export const setOnUnauthenticated = (cb: () => void) => { onUnauthenticated = cb; };
export const triggerLogout = () => { onUnauthenticated?.(); };

export async function refreshAccessToken(authClient: AuthServiceClient): Promise<boolean> {
  try {
    const r = await authClient.refresh({}); // refresh token sent via cookie
    accessToken = r.accessToken;
    return true;
  } catch {
    accessToken = null;
    return false;
  }
}

- [ ] **Step 3: Implement transport.ts**

```ts
import { createConnectTransport } from "@connectrpc/connect-web";
import { Interceptor } from "@connectrpc/connect";
import { getAccessToken, refreshAccessToken, triggerLogout } from "./auth";

const baseUrl = "/"; // same-origin in prod; vite proxies in dev

const authInterceptor: Interceptor = (next) => async (req) => {
  const tok = getAccessToken();
  if (tok) req.header.set("Authorization", `Bearer ${tok}`);
  return next(req);
};

const tenantInterceptor: Interceptor = (next) => async (req) => {
  const tenant = localStorage.getItem("active_tenant_id");
  if (tenant) req.header.set("X-Tenant-Id", tenant);
  return next(req);
};

let refreshing: Promise<boolean> | null = null;

export const transport = createConnectTransport({
  baseUrl,
  interceptors: [authInterceptor, tenantInterceptor],
  fetch: async (input, init) => {
    let resp = await fetch(input, init);
    if (resp.status === 401) {
      refreshing ??= refreshAccessToken(authClientForRefresh).finally(() => { refreshing = null; });
      const ok = await refreshing;
      if (!ok) { triggerLogout(); return resp; }
      // retry once with new token
      const retryInit = { ...init };
      const tok = getAccessToken();
      if (tok && retryInit.headers instanceof Headers) {
        retryInit.headers.set("Authorization", `Bearer ${tok}`);
      }
      resp = await fetch(input, retryInit);
    }
    return resp;
  },
});
```

export interface AppError {

  message: string;

}

export function mapConnectError(err: unknown): AppError {
  if (err instanceof ConnectError) {
    const code = (DomainCode[err.message as keyof typeof DomainCode] ?? DomainCode.UNKNOWN);

```tsx
type AuthState = { user: User | null; ready: boolean };
const AuthCtx = createContext<{ state: AuthState; login: (e: string, p: string) => Promise<void>; logout: () => Promise<void> }>(null!);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [state, setState] = useState<AuthState>({ user: null, ready: false });
  const nav = useNavigate();

  // boot — try refresh from cookie
  useEffect(() => {
    (async () => {
      const ok = await refreshAccessToken(clients.auth);
      if (ok) {
        const me = await clients.users.me({});
        setState({ user: me.user!, ready: true });
      } else {
        setState({ user: null, ready: true });
      }
    })();
    setOnUnauthenticated(() => { setState({ user: null, ready: true }); nav("/login"); });
  }, []);

  const login = async (email: string, password: string) => {
    const r = await clients.auth.login({ email, password });
    setAccessToken(r.accessToken);
    const me = await clients.users.me({});
    setState({ user: me.user!, ready: true });
    nav("/tenants");
  };

  const logout = async () => {
    try { await clients.auth.logout({}); } catch {}
    setAccessToken(null);
    localStorage.removeItem("active_tenant_id");
    setState({ user: null, ready: true });
    nav("/login");
  };

  return <AuthCtx.Provider value={{ state, login, logout }}>{children}</AuthCtx.Provider>;
}

export const useAuth = () => useContext(AuthCtx);
```

- [ ] **Step 1: useTestRunWatch.ts**

```ts
import { useEffect, useState } from "react";
import { clients } from "../api/clients";
import type { TestRunId, TestRunProgress } from "../gen/cloud/v1/testing/test_run_pb";

export function useTestRunWatch(id: TestRunId | null, sinceCursor = "") {
  const [events, setEvents] = useState<TestRunProgress[]>([]);
  const [err, setErr] = useState<string | null>(null);

  useEffect(() => {
    if (!id) return;
    const ctrl = new AbortController();
    (async () => {
      try {
        for await (const p of clients.testRuns.watchTestRun({ id, since: sinceCursor }, { signal: ctrl.signal })) {
          setEvents(prev => [...prev, p]);
        }
      } catch (e: any) {
        if (e.name !== "AbortError") setErr(String(e));
      }
    })();
    return () => ctrl.abort();
  }, [id?.value, sinceCursor]);

  return { events, err };
}
```
