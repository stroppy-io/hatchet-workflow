```go
type Harness struct {
	T            *testing.T

	Postgres     *pgcontainer.Container

	Minio        *MinioContainer
	TfStubDir    string
	cmd          *exec.Cmd

}

func New(t *testing.T, opts ...Option) *Harness {
	cfg := defaultCfg().apply(opts...)
	h := &Harness{T: t}

	h.Postgres = pgcontainer.Start(t)
	h.Valkey   = vkcontainer.Start(t)
	h.Minio    = StartMinio(t)
	h.TfStubDir = buildTfStub(t)

	binPath := buildServerBinary(t)
	configPath := writeConfig(t, ConfigOpts{
		PGDSN: h.Postgres.DSN(), ValkeyAddr: h.Valkey.Addr(),
		S3Endpoint: h.Minio.Endpoint(), S3AccessKey: h.Minio.AccessKey(), S3SecretKey: h.Minio.SecretKey(),
		TerraformBin: filepath.Join(h.TfStubDir, "terraform"),
		BootstrapAdminEmail: cfg.AdminEmail, BootstrapAdminPassword: cfg.AdminPassword,
		HTTPAddr: ":0", // ephemeral
	})

	port := pickEphemeralPort(t)
	h.cmd = exec.Command(binPath, "server", "--config", configPath, "--http-addr", fmt.Sprintf(":%d", port))
	h.cmd.Env = append(os.Environ(),
		"STROPPY_TF_STUB_LOG="+filepath.Join(t.TempDir(), "tf-stub.log"),
		"INITIAL_ADMIN_PASSWORD="+cfg.AdminPassword,
		"JWT_SECRET=test-secret",
	)
	h.cmd.Stdout, h.cmd.Stderr = newLogger(t, "server"), newLogger(t, "server")
	require.NoError(t, h.cmd.Start())
	h.ServerURL = fmt.Sprintf("http://127.0.0.1:%d", port)
	waitHealthy(t, h.ServerURL, 30*time.Second)
	t.Cleanup(func() { h.Close() })
	return h
}

func (h *Harness) Close() {
	if h.cmd != nil && h.cmd.Process != nil {
		_ = h.cmd.Process.Signal(syscall.SIGTERM)
		_ = h.cmd.Wait()
	}
	for _, f := range h.cleanupFns { f() }
}
```

```go
func TestE2E_RestartSurvival_DagRunResumes(t *testing.T) {
	h := harness.New(t)
	cli := h.AdminClient()
	tenant := h.SeedTenant("acme")

	// Launch with a slow node (mock handler that sleeps until a side-channel signal)
	tr := h.LaunchSlowDagRun(tenant)
	h.WaitForStatus(tr.Id, testingpb.TestRunStatus_TEST_RUN_STATUS_RUNNING)

	// Verify NODE_RUN is RUNNING in DB
	nrs := h.QueryNodeRuns(tr.DagRunId)
	require.True(t, anyStatus(nrs, systempb.NodeRunStatus_NODE_RUN_STATUS_RUNNING))

	// Kill server (SIGKILL) and restart
	h.KillServerHard()
	h.RestartServer()

	// Recovery sweep should have reset RUNNING → READY+attempt+1
	require.Eventually(t, func() bool {
		nrs := h.QueryNodeRuns(tr.DagRunId)
		for _, n := range nrs {
			if n.Status == systempb.NodeRunStatus_NODE_RUN_STATUS_READY && n.Attempt >= 1 {
				return true
			}
		}
		return false
	}, 30*time.Second, 250*time.Millisecond)

	// Unblock the slow node; run reaches SUCCEEDED
	h.UnblockSlowNode()
	require.Eventually(t, func() bool {
		got, _ := cli.TestRuns.GetTestRun(h.Ctx(), tr.Id)
		return got.TestRun.Status == testingpb.TestRunStatus_TEST_RUN_STATUS_SUCCEEDED
	}, 2*time.Minute, 500*time.Millisecond)
}
```

```go
func TestE2E_MultiTenantIsolation(t *testing.T) {
	h := harness.New(t)
	a, _ := h.Admin().CreateTenant(...) // tenant A + lead-a
	b, _ := h.Admin().CreateTenant(...) // tenant B + lead-b
	leadA := h.LoginAs("a@x.test", ...); leadA.UseTenant(a.Id)
	leadB := h.LoginAs("b@x.test", ...); leadB.UseTenant(b.Id)

	trA := leadA.SimpleTestRun(t)

	// B cannot read A's run
	_, err := leadB.TestRuns.GetTestRun(h.Ctx(), trA.Id)
	require.ErrorContains(t, err, "PERMISSION_DENIED")

	// B's list does not include A's run
	listB, _ := leadB.TestRuns.ListTestRuns(h.Ctx(), b.Id)
	for _, r := range listB.Items {
		require.NotEqual(t, trA.Id.Value, r.Id.Value)

	_, err = leadB.TestRuns.CreateTestRun(h.Ctx(), &testingpb.CreateTestRunRequest{TenantId: a.Id})

	// A's webhook does not fire on B's TestRunCompleted
	hookA := h.MockHook()
	leadA.Webhooks.CreateWebhook(h.Ctx(), webhookFor(a, hookA.URL))
	leadB.RunTestRunToCompletion(t)
	require.Eventually(t, func() bool { return !hookA.Received() }, 5*time.Second, 100*time.Millisecond)
```

```go
//go:build perf

func TestPerf_ConcurrentLaunchLatency_P95(t *testing.T) {
	h := harness.New(t, harness.WithMockNodeHandlersFast())
	cli := h.AdminClient()
	tenant := h.SeedTenant("perf")
	tpl, _ := cli.TestRunTemplates.CreateTestRunTemplate(h.Ctx(), fixtures.TrivialTemplateReq(tenant))

	const N = 100
	latencies := make([]time.Duration, N)
	var wg sync.WaitGroup; wg.Add(N)
	for i := 0; i < N; i++ {
		go func(i int) {
			defer wg.Done()
			start := time.Now()
			tr, err := cli.TestRuns.InstantiateTestRun(h.Ctx(), tpl.Id)
			require.NoError(t, err)
			_, err = cli.TestRuns.LaunchTestRun(h.Ctx(), tr.Id)
			require.NoError(t, err)
			latencies[i] = time.Since(start)
		}(i)
	}
	wg.Wait()

	sort.Slice(latencies, func(i, j int) bool { return latencies[i] < latencies[j] })
	p50 := latencies[N/2]; p95 := latencies[N*95/100]; p99 := latencies[N*99/100]
	t.Logf("p50=%s p95=%s p99=%s", p50, p95, p99)
	require.Less(t, p95, 500*time.Millisecond, "p95 launch latency budget")
}
```

```markdown
1. Build: `make build` → `./bin/stroppy-cloud`
2. Provision Postgres + Valkey + S3 (compatible) + (optional) VictoriaMetrics.
```

```markdown
# Restart

Single binary, blanket reset on boot:
- All `node_runs.RUNNING` → `READY` (attempt+1) or `FAILED` (attempt>=retry_max)
- All `webhook_deliveries.IN_FLIGHT` → `PENDING`
- Stale agents (last_seen > 60s) → `UNKNOWN`
- DagRun statuses recomputed from aggregate

Steps:
1. `SIGTERM` (graceful, 30s drain).
2. Wait for process exit.
3. Start fresh process. Recovery runs before NodeWorkers + Scheduler.
4. Watch logs for `recovery: reset N node_runs, N deliveries`.
```

- [ ] **Step 3: troubleshooting.md**

```markdown
# Troubleshooting

## Login returns UNAUTHENTICATED
- Check JWT_SECRET matches between restarts.
- Inspect refresh-token cookie; clear if persistent invalidation.

## NodeRun stuck in READY
- Check `workers.node_workers` > 0 and pool not deadlocked.
- Look for `claim returned 0 rows` in logs (no work) vs. `claim failed` (DB).

## Webhook DEAD before reaching limit
- Inspect `webhook_deliveries.last_error`. 4xx-non-retryable returns DEAD on first try.

## Agent stuck UNKNOWN
- Verify reverse proxy allows 25s long-poll. Some Nginx defaults idle-timeout at 60s — OK.
- Check `agents.last_seen_at`; if old, the agent process likely died.
```
