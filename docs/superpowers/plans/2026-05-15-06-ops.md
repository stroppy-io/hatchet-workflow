# Ops Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the operational subsystem — `WebhookService` + delivery outbox worker, `QuotaService` + enforcement port for testing services, `BinaryCacheService.Resolve` (real implementation backing the agent verb from Plan 05).

**Architecture:** Three services in `internal/domain/services/ops/`. Webhooks: in-memory `eventing.Bus` subscriber (set up at server bootstrap) pattern-matches incoming events against `webhooks.subscribed_events`, `INSERT`s into `webhook_deliveries`. A separate `webhookworker` goroutine drains `state=PENDING AND next_attempt_at <= now()` rows, POSTs to subscriber URL with HMAC-SHA256 signature header, and on failure schedules an exponential backoff retry. Quota: simple atomic counter table `quota_counters(tenant_id, resource_kind, used, limit_value, period_start)`; a `QuotaPort` interface is implemented in `ops` and consumed by `testing.TestRunService.LaunchTestRun` to check + increment before launch. BinaryCache: ratel-backed `binary_artifacts(package_id, format, url, sha256, extract_to, ttl)` table populated by admin upload flow (Plan 07); `Resolve(packageID)` reads the row and returns a presigned-S3 URL (or pass-through public URL) with refreshed TTL.

**Tech Stack:** Standard library `net/http` for outbound webhook POSTs, `crypto/hmac` for signatures, `internal/infrastructure/s3` for presign, ratel repos, `pgtx`, in-memory `eventing.Bus`.

**Prerequisites:** Plans 01-05 merged. The `eventing.Bus` exists and publishes `TestRunCompleted`, `DagRunCompleted`, `AgentRegistered`, etc. (defined in Plan 01).

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/hex"
)

func Sign(payload, secret []byte) string {
	h := hmac.New(sha256.New, secret)
	h.Write(payload)
	return "sha256=" + hex.EncodeToString(h.Sum(nil))
}

```go
type Worker struct {
	delivRepo *repository.ProtoRepository[opspb.WebhookDeliveryAlias, ...]
	whRepo    *repository.ProtoRepository[opspb.WebhookAlias, ...]
	txManager pgtx.TxManager
	httpc     *http.Client
	concurrency int
	tick       time.Duration
	maxAttempts int
}

func (w *Worker) Run(ctx context.Context) {
	sem := make(chan struct{}, w.concurrency)
	tick := time.NewTicker(w.tick); defer tick.Stop()
	for {
		select { case <-ctx.Done(): return; case <-tick.C: }
		claims, err := w.claimBatch(ctx, w.concurrency)
		if err != nil { continue }
		for _, d := range claims {
			sem <- struct{}{}
			go func(d *opspb.WebhookDelivery) {
				defer func() { <-sem }()
				w.deliver(ctx, d)
			}(d)
		}
	}
}

func (w *Worker) claimBatch(ctx context.Context, n int) ([]*opspb.WebhookDelivery, error) {
	return pgtx.WithSerializableRet(ctx, w.txManager, func(ctx context.Context) ([]*opspb.WebhookDelivery, error) {
		rows, err := w.delivRepo.FindAllRaw(ctx, claimSQL, n) // SELECT ... WHERE state='PENDING' AND next_attempt_at<=now() ORDER BY next_attempt_at LIMIT $1 FOR UPDATE SKIP LOCKED
		if err != nil { return nil, err }
		for _, r := range rows {
			r.State = opspb.WebhookDeliveryState_WEBHOOK_DELIVERY_STATE_IN_FLIGHT
			r.Attempts++
			_ = w.delivRepo.Update(ctx, r)
		}
		return rows, nil
	})
}

func (w *Worker) deliver(ctx context.Context, d *opspb.WebhookDelivery) {
	wh, err := w.whRepo.FindByID(ctx, d.WebhookId.GetValue())
	if err != nil { w.markDead(ctx, d, "webhook_missing"); return }
	req, _ := http.NewRequestWithContext(ctx, "POST", wh.Url, bytes.NewReader(d.Payload))
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("X-Stroppy-Signature", Sign(d.Payload, []byte(wh.Secret)))
	req.Header.Set("X-Stroppy-Delivery-Id", d.Id.GetValue())

	resp, err := w.httpc.Do(req)
	switch {
	case err != nil:
		w.scheduleRetry(ctx, d, err.Error())
	case resp.StatusCode >= 200 && resp.StatusCode < 300:
		w.markDelivered(ctx, d)
	case resp.StatusCode >= 400 && resp.StatusCode < 500 && resp.StatusCode != 408 && resp.StatusCode != 429:
		w.markDead(ctx, d, fmt.Sprintf("http_%d", resp.StatusCode))
	default:
		w.scheduleRetry(ctx, d, fmt.Sprintf("http_%d", resp.StatusCode))
	}
}

func (w *Worker) scheduleRetry(ctx context.Context, d *opspb.WebhookDelivery, lastErr string) {
	if d.Attempts >= int32(w.maxAttempts) { w.markDead(ctx, d, lastErr); return }
	backoff := time.Duration(1<<d.Attempts) * 30 * time.Second
	if backoff > 1*time.Hour { backoff = 1 * time.Hour }
	d.State = opspb.WebhookDeliveryState_WEBHOOK_DELIVERY_STATE_PENDING
	d.NextAttemptAt = timestamppb.New(time.Now().Add(backoff))
	d.LastError = lastErr
	_ = w.delivRepo.Update(ctx, d)
}
```

type BinaryCacheService struct {
	pkgRepo *repository.ProtoRepository[catalogpb.PackageAlias, ...]
	s3      S3Presigner
	iam     IamPort
}

func (s *BinaryCacheService) Resolve(ctx context.Context, req *agentpb.ResolveArtifactRequest) (*agentpb.ResolveArtifactResponse, error) {
	agentID, err := s.iam.AgentFromCtx(ctx)
	if err != nil { return nil, err }
	ag, err := s.iam.LookupAgent(ctx, agentID)
	if err != nil { return nil, err }
	pkg, err := s.pkgRepo.FindByID(ctx, req.PackageId.GetValue())
	if err != nil { return nil, domainerr.FromPg(err) }
	if pkg.TenantId.GetValue() != ag.TenantId.GetValue() && !pkg.Public {
		return nil, domainerr.E(errorspb.Code_PERMISSION_DENIED)
	}
	url, err := s.s3.Presign(ctx, pkg.S3Bucket, pkg.S3Key, 15*time.Minute)
	if err != nil { return nil, err }
	return &agentpb.ResolveArtifactResponse{
		Url:       url,
		Sha256:    pkg.Sha256,
		Format:    pkg.ArtifactFormat,
		ExtractTo: pkg.ExtractTo,
		Name:      pkg.Name,
	}, nil
}

func TestE2E_TestRunCompleted_FiresWebhook(t *testing.T) {
	got := make(chan []byte, 1)
	mockHook := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {

	}))

	srv := e2e.StartServer(t, e2e.WithMockNodeHandlers())
	defer srv.Close()
	cli := srv.LoginAsAdmin()
	tenant := cli.CurrentTenant()
	_, _ = cli.Webhooks.CreateWebhook(srv.Ctx(), &opspb.CreateWebhookRequest{
		TenantId: tenant.Id, Url: mockHook.URL, Secret: "shh",
		SubscribedEvents: []opspb.WebhookEventKind{opspb.WebhookEventKind_WEBHOOK_EVENT_KIND_TEST_RUN_COMPLETED},
	})
	tpl, _ := cli.Templates.CreateTestRunTemplate(srv.Ctx(), &testingpb.CreateTestRunTemplateRequest{...})
	tr, _ := cli.Runs.InstantiateTestRun(srv.Ctx(), tpl.Id)
	_, _ = cli.Runs.LaunchTestRun(srv.Ctx(), tr.Id)

	select {
	case body := <-got:
		require.Contains(t, string(body), tr.Id.Value)
	case <-time.After(60 * time.Second):

	}
}
